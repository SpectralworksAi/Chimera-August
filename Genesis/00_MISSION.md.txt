def _clip01(x: float) -> float:
    return 0.0 if x < 0.0 else 1.0 if x > 1.0 else x


def resonance(S: Dict, M: Dict) -> float:
    v = np.array([_clip01(S["valence"]), _clip01(S["arousal"]), _clip01(S["coherence"])])
    m = np.array([_clip01(M["valence"]), _clip01(M["arousal"]), _clip01(M["coherence"])])
    dist = np.linalg.norm(v - m) / np.sqrt(3)
    return _clip01(1.0 - dist)


def entropy(M: Dict) -> float:
    vals = [_clip01(M["valence"]), _clip01(M["arousal"]), 1.0 - _clip01(M["coherence"])]
    return float(np.std(vals))


def update_state(S_t: Dict, M_t: Dict) -> Tuple[Dict, float, float, float, str]:
    R = resonance(S_t, M_t)
    E = entropy(M_t)
    if R < 0.4 or E > 0.5:
        alpha, trigger = 0.3, "collapse"
    elif R < 0.7 or E > 0.25:
